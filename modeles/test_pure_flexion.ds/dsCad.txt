# =============================================================================
# Test pure flexion beton -- 3 lits de 8HA32
#
# b = 0.8m, h = 0.8m, L = 5.0m
# Origine z a h/2 = 0.4m (z positif vers le haut = cote tendu)
# Lit 1 : z = +0.334m (6.6cm du bord haut),  HA1..HA8
# Lit 2 : z = +0.302m (9.8cm du bord haut),  HA9..HA16
# Lit 3 : z = +0.238m (16.2cm du bord haut), HA17..HA24
# Centroide acier : z_moy = 0.291m (10.9cm du bord haut)
# d = h/2 + z_moy = 0.4 + 0.291 = 0.691m
# 8 barres par lit : y = -0.35, -0.25, -0.15, -0.05, +0.05, +0.15, +0.25, +0.35
# =============================================================================

# --- Materiaux ---
fy    = 550.0000000000
fc    = 48.0000000000
ft1    = 0.1
ft2 = 3.5
E     = 35.0  # module Young (GPa)

# --- Resistances avec coefficient de sécurité ---
gamma_s = 1.0
gamma_c = 1.0
fyd = fy/gamma_s
fcd = fc/gamma_c

# --- Geometrie ---
b  = 0.8    # largeur (m)
h  = 0.8    # hauteur (m)
L  = 5.0    # longueur (m)
L2 = 1.0
phi = 32.0  # diametre des aciers (mm)


BLOCK('Block1', L-L2, 'RECTANGLE', {'b':b, 'h':h}, 0,
      POINT(0, 0, 0),
      Plus_XLoc, 'XAxis',
      Plus_ZLoc, 'ZAxis',
      COMPRESSIVE_STRENGTH=str(fcd),
      TENSILE_STRENGTH=str(ft1),
      YOUNG_MODULUS=str(E),
      POISSON_RATIO='0.1')
BLOCK('Block2',L2 , 'RECTANGLE', {'b':b, 'h':h}, 0,
      POINT(L-L2, 0, 0),
      Plus_XLoc, 'XAxis',
      Plus_ZLoc, 'ZAxis',
      COMPRESSIVE_STRENGTH=str(fcd),
      TENSILE_STRENGTH=str(ft2),
      YOUNG_MODULUS=str(E),
      POISSON_RATIO='0.1')
# Connection glued
CONNECT(["Block1"],["Block2"],GLUED)

# Encastrement face x=0, force ponctuelle descendante face x=L (selon Z negatif)
BOUNDARIES(
    BOUNDARY('Boundary0', 'Block1:f1', POINT(0, 0, 0),
             X=VECTOR(-1, 0, 0), Y=VECTOR(0, -1, 0)),
    BOUNDARY('Boundary1', 'Block2:f2', POINT(L, 0, 0),
             X=VECTOR(1, 0, 0), Y=VECTOR(0, 1, 0))
)

pts0.append(POINT(L, -0.35, 0.334))
